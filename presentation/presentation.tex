\section{Forschungsfragen \& Hypothesen}

\begin{frame}
\frametitle{Drei Hauptforschungsfragen}
\begin{enumerate}
\item[\textcolor{LMUBlue}{\textbf{RQ1}}] \researchq{To what extent does a game-based approach on a public display help deepfake recognition?}

\item[\textcolor{LMUGreen}{\textbf{RQ2}}] \researchq{How do presented detection strategies contribute to raising awareness?}

\item[\textcolor{LMUGray}{\textbf{RQ3}}] \researchq{How do users perceive public displays for recognizing deepfake videos?}
\end{enumerate}
\end{frame}

\begin{frame}
\frametitle{Zentrale Hypothesen (Auswahl)}
\begin{block}{H1b: Gamification erhöht Lernmotivation}
Die spielerische Form erhöht signifikant die Motivation zum Lernen der Deepfake-Erkennung
\end{block}

\begin{block}{H2b: Strategien erhöhen Selbstvertrauen}
Die Selbsteinschätzung der Deepfake-Erkennungsfähigkeit ist nach dem Spiel signifikant höher als vorher
\end{block}

\begin{block}{H3b: Public Display erreicht diverse Nutzergruppen}
Nutzer mit unterschiedlicher Public Display Erfahrung nutzen das System gleichermaßen
\end{block}
\end{frame}
